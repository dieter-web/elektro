### Vorschlag: Schaltung mit Matrizen (Knotenleitwertmethode) berechnen

Kurze Antwort: ja — statt Schritt-für-Schritt-Reihen-/Parallelsummen kannst du das ganze Netzwerk als lineares Gleichungssystem G·V = I (Knotenleitwertmatrix) aufbauen und lösen. Dann liest du die Spannungen an den Knoten aus und berechnest daraus U7 und I7. Unten findest du ein kompaktes, wiederverwendbares Muster in JavaScript, das du in deinem Modul einbauen kannst.

---

### Idee in einem Satz
Baue aus den Widerständen eine Liste von Kanten (nodeA, nodeB, R), konstruiere die Leitwertmatrix G (G_ii = Sum(g_ik), G_ij = -g_ij), setze die Knotenstromquelle/-spannung (geschlossen über Quelle als Einspannung an einem Knoten gegenüber Referenz), löse V = G^{-1}·I und berechne gesuchte Spannungen und Ströme.

---

### Schritte
1. Wähle Knotennummern; einer ist Referenz (Masse = 0).  
2. Erzeuge Kanten: jedes R zwischen zwei Knoten als Leitwert g = 1/R.  
3. Fülle G-Matrix (n x n, ohne Referenzknoten) und I-Vektor.  
4. Löse das Gleichungssystem (z.B. mit mathjs oder eigenem Gauss).  
5. Berechne U7 als Differenz der Knoten, I7 = U über R7 / R7.  
6. Rechne Prozentwerte wie vorher.

```js
// Benötigt nur Standard-JS; wenn du mathjs hast, kannst du math.inv/solve verwenden.
// Hier: einfacher Gauss-Löser für reelle Matrizen.

function solveLinear(A, b) {
  const n = A.length;
  // deep copy
  const M = A.map((row, i) => row.concat(b[i]));
  // Gauss-Jordan
  for (let i = 0; i < n; i++) {
    // pivot
    let max = i;
    for (let k = i + 1; k < n; k++) if (Math.abs(M[k][i]) > Math.abs(M[max][i])) max = k;
    if (Math.abs(M[max][i]) < 1e-12) throw new Error('Singuläre Matrix');
    [M[i], M[max]] = [M[max], M[i]];
    const piv = M[i][i];
    for (let j = i; j <= n; j++) M[i][j] /= piv;
    for (let r = 0; r < n; r++) if (r !== i) {
      const fac = M[r][i];
      for (let c = i; c <= n; c++) M[r][c] -= fac * M[i][c];
    }
  }
  return M.map(row => row[n]);
}

// Netzwerk-Helper: Knoten 0 ist Referenz (Masse).
function computeByConductance(nodesCount, resistors, voltageSources) {
  // resistors: [{n1, n2, R}]  n1,n2 integers, 0..N
  // voltageSources: [{plusNode, minusNode, V}] treat as ideal voltage between nodes;
  // for single ideal source between reference (0) and node k, we can place known node voltage.
  // This helper supports only sources where one terminal is reference (0).
  // nodesCount includes reference; we'll build for unknown nodes 1..N-1
  const N = nodesCount; // total nodes including ref 0
  const unknownCount = N - 1; // nodes with unknown voltage: 1..N-1
  const G = Array.from({length: unknownCount}, () => Array(unknownCount).fill(0));
  const I = Array(unknownCount).fill(0);

  // Mark nodes with fixed voltages (due to ideal source to ref)
  const fixedVoltages = {}; // node -> voltage
  (voltageSources || []).forEach(s => {
    if (s.minusNode !== 0 && s.plusNode !== 0) {
      throw new Error('Dieses einfache helper unterstützt nur Quellen gegen Referenz (node 0).');
    }
    const node = s.plusNode === 0 ? s.minusNode : s.plusNode;
    const V = s.plusNode === node ? s.V : -s.V; // ensure V at node = plusNode - minusNode
    fixedVoltages[node] = V;
  });

  // Add conductances
  for (const {n1, n2, R} of resistors) {
    const g = 1 / R;
    // four Fälle: beide unknown, eine fixed, beide fixed
    const isFixed1 = (n1 === 0) || (fixedVoltages.hasOwnProperty(n1));
    const isFixed2 = (n2 === 0) || (fixedVoltages.hasOwnProperty(n2));
    const val1 = fixedVoltages[n1] ?? 0;
    const val2 = fixedVoltages[n2] ?? 0;

    if (!isFixed1 && !isFixed2) {
      const i = n1 - 1, j = n2 - 1;
      G[i][i] += g;
      G[j][j] += g;
      G[i][j] -= g;
      G[j][i] -= g;
    } else if (!isFixed1 && isFixed2) {
      const i = n1 - 1;
      G[i][i] += g;
      I[i] += g * val2; // move known voltage to RHS
    } else if (isFixed1 && !isFixed2) {
      const j = n2 - 1;
      G[j][j] += g;
      I[j] += g * val1;
    } // else both fixed -> kein Beitrag zur Unbekanntenmatrix
  }

  // solve for unknown node voltages
  const V_unknown = solveLinear(G, I);
  const V = Array(N).fill(0);
  // node 0 is 0 (reference)
  for (let k = 1; k < N; k++) V[k] = fixedVoltages.hasOwnProperty(k) ? fixedVoltages[k] : V_unknown[k - 1];
  return V;
}

// Beispiel: Zuordnung der Knoten für deine Schaltung
// Du musst die Knotenzuordnung passend zur Schaltbild-Topologie setzen.
// Angenommen (Beispiel-Topologie):
// Node 0 = Masse (unterer Schienen)
// Node 1 = Knoten zwischen R1 und Rest (nach R1)
// Node 2 = Knoten, von dem R2 geht
// Node 3 = Knoten für den R6-R7-Zweig etc.
// (Diese Zuordnung ist nur ein Template — an dein Bild anpassen!)

function exampleCompute(input) {
  // parse numeric values (falls input als "30ohm" kommt, strip Einheit)
  const parse = v => Number(String(v).replace(/[^\d.-]/g, ''));
  const R = {};
  for (let i=1;i<=7;i++) R[i] = parse(input['R'+i]);
  const U = parse(input.U);

  // Hier definierst du die Knotenbelegung für jeden Widerstand:
  // resistors = [{n1,n2,R}] mit n in 0..N-1
  // WICHTIG: passe die Knotenzuordnung an dein Schaltbild an!
  const resistors = [
    // Beispielzuordnung (nur Template!)
    {n1: 0, n2: 1, R: R[1]}, // R1 zwischen 0 und 1
    {n1: 1, n2: 2, R: R[2]}, // R2 zwischen 1 und 2
    {n1: 2, n2: 3, R: R[3]}, // R3 zwischen 2 und 3
    {n1: 3, n2: 1, R: R[4]}, // R4 zwischen 3 und 1
    {n1: 2, n2: 0, R: R[5]}, // R5 zwischen 2 und 0
    {n1: 3, n2: 4, R: R[6]}, // R6 zwischen 3 und 4
    {n1: 4, n2: 0, R: R[7]}, // R7 zwischen 4 und 0
  ];
  // nodesCount = highest node index + 1
  const nodesCount = 5; // 0..4 in diesem Template

  // Voltage source: U between node 0 (minus) and node 1 (plus) => V(node1)=U
  const sources = [{ plusNode: 1, minusNode: 0, V: U }];

  const V = computeByConductance(nodesCount, resistors, sources);

  // Beispiel: U7 ist Spannung über R7: zwischen node 4 und 0
  const U7 = V[4] - V[0];
  const I7 = U7 / R[7];

  // Gesamtstrom aus Quelle I1 = Sum-Ströme aus Node1 in Richtung Referenz
  // alternativ: I1 = (U - V_at_node_connected_to_ref_through_R1)/R1 ... abhängig von Topo
  // Hier: bestimme Gesamtstrom als Summe der Ströme an Knoten 1 in Richtung angeschlossener Leiter
  // Einfacher: Gesamtstrom aus Quelle gleich Summe der Ströme durch Widerstände, die Node1 mit anderen Knoten verbinden:
  function currentThrough(nA, nB, Rv, Varr){ return (Varr[nA]-Varr[nB]) / Rv; }
  // Beispiel: Gesamtstrom aus Node1 zur Referenz (Summe Ströme von node1 zu allen Nachbarn)
  const neighbors = resistors.filter(r=> r.n1===1 || r.n2===1);
  let I_total_from_node1 = 0;
  for (const r of neighbors) {
    const a = r.n1, b = r.n2;
    const sign = (a===1) ? 1 : -1; // if n1==1 current = (V1-Vb)/R ; if n2==1 then (V1-Va)/R
    const other = a===1 ? b : a;
    I_total_from_node1 += sign * ((V[1] - V[other]) / r.R);
  }

  const U7_percent = (U7 / U) * 100;
  const I7_percent = (I7 / I_total_from_node1) * 100;

  return { V, U7, U7_percent, I7, I7_percent, I_total_from_node1 };
}
```

```js
// solveLinear und computeByConductance wie zuvor definiert (Gauss-Jordan + Helfer).
// Hier nur der Teil, den du ersetzt/erweiterst: Parsing, Remapping, Aufruf.

function parseNum(v) { return Number(String(v).replace(/[^\d.-]/g, '')); }

// Deine Zuordnung (aus der Nachricht)
const resistors_input = [
  {n1: 2, n2: 1, Rkey: 1}, // R1 zwischen 2 und 1
  {n1: 3, n2: 2, Rkey: 2}, // R2 zwischen 3 und 2
  {n1: 2, n2: 4, Rkey: 3}, // R3 zwischen 2 und 4
  {n1: 5, n2: 4, Rkey: 4}, // R4 zwischen 5 und 4
  {n1: 5, n2: 3, Rkey: 5}, // R5 zwischen 5 und 3
  {n1: 6, n2: 4, Rkey: 6}, // R6 zwischen 6 und 4
  {n1: 6, n2: 5, Rkey: 7}, // R7 zwischen 6 und 5
];

// Hauptfunktion: input erwartet Felder R1..R7 und U
function computeFromMapping(input) {
  const R = {};
  for (let i = 1; i <= 7; i++) R[i] = parseNum(input['R' + i]);
  const U = parseNum(input.U);

  // Remap: set node1 as reference (0). newNode = oldNode - 1
  const resistors = resistors_input.map(r => ({
    n1: r.n1 - 1,
    n2: r.n2 - 1,
    R: R[r.Rkey]
  }));

  // nodesCount = max node index + 1
  const maxNode = Math.max(...resistors.flatMap(r => [r.n1, r.n2]));
  const nodesCount = maxNode + 1;

  // Spannungsquelle: plus an alten Knoten 2, minus an alten Knoten1 (Referenz)
  // Nach Remap: plusNode = 1, minusNode = 0
  const sources = [{ plusNode: 1, minusNode: 0, V: U }];

  const V = computeByConductance(nodesCount, resistors, sources);

  // R7 liegt zwischen alten Knoten 6 und 5 -> nach Remap 5 und 4
  const nodeA_R7 = 6 - 1; // 5
  const nodeB_R7 = 5 - 1; // 4
  const U7 = V[nodeA_R7] - V[nodeB_R7];
  const I7 = U7 / R[7];

  // Bestimme Gesamtstrom aus der Quelle (Summe Ströme vom plus-Knoten 1 zu Nachbarn)
  const plusNode = 1;
  const neighbors = resistors.filter(r => r.n1 === plusNode || r.n2 === plusNode);
  let I_total_from_plus = 0;
  for (const r of neighbors) {
    const other = (r.n1 === plusNode) ? r.n2 : r.n1;
    I_total_from_plus += (V[plusNode] - V[other]) / r.R;
  }

  const U7_percent = (U7 / U) * 100;
  const I7_percent = (I7 / I_total_from_plus) * 100;

  return {
    nodesCount,
    V,
    U7,
    U7_percent,
    I7,
    I7_percent,
    I_total_from_plus
  };
}

// Beispielaufruf mit Zahlen (Einheiten-entfernt oder als "30ohm")
const inputExample = { R1: '30', R2: '70', R3: '20', R4: '160', R5: '40', R6: '80', R7: '120', U: '24' };
const result = computeFromMapping(inputExample);
console.log(result);
```

### Hinweise zur Interpretation der Ergebnisse
- V enthält die Knotenpotentiale relativ zur Referenz (alter Knoten 1).
- U7 ist die Spannung über R7 (positiv wenn Knoten6 gegenüber Knoten5 höheres Potential hat).
- I7 ist der Strom durch R7 (gerichtet von Knoten6 nach Knoten5 entsprechend U7).
- I_total_from_plus ist der Gesamtstrom, den die Quelle liefert (entsprechend deinem I1). I7_percent ist Prozent von I1.
