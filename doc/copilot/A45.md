### Substitutionslösung des Gleichungssystems

Gegeben:
\[
\begin{aligned}
(1)\quad& I_1 + I_2 - I_3 - I_4 = 0 \\
(2)\quad& -R_{i1} I_1 + R_{i2} I_2 = U_2 - U_1 \\
(3)\quad& -R_{i2} I_2 - R_3 I_3 = -U_2 \\
(4)\quad& R_3 I_3 - R_4 I_4 = 0
\end{aligned}
\]

---

#### Schritt 1: Direkt ausdrücken aus (4), (3), (2)
- **Aus (4):** 
\[
I_4 = \frac{R_3}{R_4}\, I_3
\]

- **Aus (3):**
\[
R_{i2} I_2 + R_3 I_3 = U_2 \quad\Rightarrow\quad I_2 = \frac{U_2 - R_3 I_3}{R_{i2}}
\]

- **Aus (2):**
\[
-R_{i1} I_1 + R_{i2} I_2 = U_2 - U_1 \quad\Rightarrow\quad I_1 = \frac{R_{i2} I_2 - (U_2 - U_1)}{R_{i1}}
\]
Einsetzen von \(I_2\) liefert die nützliche Vereinfachung:
\[
I_1 = \frac{U_1 - R_3 I_3}{R_{i1}}
\]

---

#### Schritt 2: In (1) einsetzen und nach \(I_3\) lösen
In (1) mit obigen Ausdrücken:
\[
\frac{U_1 - R_3 I_3}{R_{i1}} + \frac{U_2 - R_3 I_3}{R_{i2}} - I_3 - \frac{R_3}{R_4} I_3 = 0
\]
Gruppieren nach \(I_3\):
\[
\frac{U_1}{R_{i1}} + \frac{U_2}{R_{i2}} - I_3\left(\frac{R_3}{R_{i1}} + \frac{R_3}{R_{i2}} + 1 + \frac{R_3}{R_4}\right) = 0
\]
Damit:
\[
I_3 = \frac{\dfrac{U_1}{R_{i1}} + \dfrac{U_2}{R_{i2}}}{\dfrac{R_3}{R_{i1}} + \dfrac{R_3}{R_{i2}} + 1 + \dfrac{R_3}{R_4}}
\]

---

#### Schritt 3: Rücksubstitution für \(I_4, I_2, I_1\)
- **Strom \(I_4\):**
\[
I_4 = \frac{R_3}{R_4}\, I_3
\]

- **Strom \(I_2\):**
\[
I_2 = \frac{U_2 - R_3 I_3}{R_{i2}}
\]

- **Strom \(I_1\):**
\[
I_1 = \frac{U_1 - R_3 I_3}{R_{i1}}
\]

---

#### Kompakte Gesamtlösung
\[
\boxed{
\begin{aligned}
I_3 &= \frac{\dfrac{U_1}{R_{i1}} + \dfrac{U_2}{R_{i2}}}{\dfrac{R_3}{R_{i1}} + \dfrac{R_3}{R_{i2}} + 1 + \dfrac{R_3}{R_4}} \\
I_4 &= \frac{R_3}{R_4}\, I_3 \\
I_2 &= \frac{U_2 - R_3 I_3}{R_{i2}} \\
I_1 &= \frac{U_1 - R_3 I_3}{R_{i1}}
\end{aligned}
}
\]

  const denominator = R3 / Ri1 + R3 / Ri2 + 1 + R3 / R4;
  const I3 = numerator / denominator;

const { create, all } = require('mathjs');
const math = create(all);

// Funktion zur Berechnung der Ströme
function solveCurrents({ Ri1, Ri2, R3, R4, U1, U2 }) {
  // Eingaben als mathjs-Units
  const Ri1_u = math.unit(Ri1, 'ohm');
  const Ri2_u = math.unit(Ri2, 'ohm');
  const R3_u  = math.unit(R3, 'ohm');
  const R4_u  = math.unit(R4, 'ohm');
  const U1_u  = math.unit(U1, 'volt');
  const U2_u  = math.unit(U2, 'volt');

  // Schritt 1: I3 berechnen
  const numerator   = math.add(math.divide(U1_u, Ri1_u), math.divide(U2_u, Ri2_u));
  const denominator = math.add(
    math.add(math.divide(R3_u, Ri1_u), math.divide(R3_u, Ri2_u)),
    math.add(1, math.divide(R3_u, R4_u))
  );
  const I3 = math.divide(numerator, denominator);

  // Schritt 2: Rücksubstitution
  const I4 = math.multiply(math.divide(R3_u, R4_u), I3);
  const I2 = math.divide(math.subtract(U2_u, math.multiply(R3_u, I3)), Ri2_u);
  const I1 = math.divide(math.subtract(U1_u, math.multiply(R3_u, I3)), Ri1_u);

  // Ergebnisse als Ampere zurückgeben
  return {
    I1: I1.to('A').toString(),
    I2: I2.to('A').toString(),
    I3: I3.to('A').toString(),
    I4: I4.to('A').toString()
  };
}

// Beispielaufruf:
const result = solveCurrents({

  Ri2: 20,   // Ohm
  R3: 30,    // Ohm
  R4: 40,    // Ohm
  U1: 5,     // Volt
  U2: 12     // Volt
});

console.log(result);
// -> { I1: '0.25 A', I2: '0.35 A', I3: '0.30 A', I4: '0.225 A' }

module.exports = solveCurrents;
