// currents.js
function solveCurrents({ Ri1, Ri2, R3, R4, U1, U2 }) {
  // Schritt 1: I3 berechnen
  const numerator = U1 / Ri1 + U2 / Ri2;
  const denominator = R3 / Ri1 + R3 / Ri2 + 1 + R3 / R4;
  const I3 = numerator / denominator;

  // Schritt 2: Rücksubstitution
  const I4 = (R3 / R4) * I3;
  const I2 = (U2 - R3 * I3) / Ri2;
  const I1 = (U1 - R3 * I3) / Ri1;

  // Ergebnis als Objekt zurückgeben
  return { I1, I2, I3, I4 };
}

// Beispielaufruf:
const result = solveCurrents({
  Ri1: 10,
  Ri2: 20,
  R3: 30,
  R4: 40,
  U1: 5,
  U2: 12
});

console.log(result);
// -> { I1: ..., I2: ..., I3: ..., I4: ... }

module.exports = solveCurrents;

Node.js mit mathjs

// currents-mathjs.js
const { create, all } = require('mathjs');
const math = create(all);

// Funktion zur Berechnung der Ströme
function solveCurrents({ Ri1, Ri2, R3, R4, U1, U2 }) {
  // Eingaben als mathjs-Units
  const Ri1_u = math.unit(Ri1, 'ohm');
  const Ri2_u = math.unit(Ri2, 'ohm');
  const R3_u  = math.unit(R3, 'ohm');
  const R4_u  = math.unit(R4, 'ohm');
  const U1_u  = math.unit(U1, 'volt');
  const U2_u  = math.unit(U2, 'volt');

  // Schritt 1: I3 berechnen
  const numerator   = math.add(math.divide(U1_u, Ri1_u), math.divide(U2_u, Ri2_u));
  const denominator = math.add(
    math.add(math.divide(R3_u, Ri1_u), math.divide(R3_u, Ri2_u)),
    math.add(1, math.divide(R3_u, R4_u))
  );
  const I3 = math.divide(numerator, denominator);

  // Schritt 2: Rücksubstitution
  const I4 = math.multiply(math.divide(R3_u, R4_u), I3);
  const I2 = math.divide(math.subtract(U2_u, math.multiply(R3_u, I3)), Ri2_u);
  const I1 = math.divide(math.subtract(U1_u, math.multiply(R3_u, I3)), Ri1_u);

  // Ergebnisse als Ampere zurückgeben
  return {
    I1: I1.to('A').toString(),
    I2: I2.to('A').toString(),
    I3: I3.to('A').toString(),
    I4: I4.to('A').toString()
  };
}

// Beispielaufruf:
const result = solveCurrents({
  Ri1: 10,   // Ohm
  Ri2: 20,   // Ohm
  R3: 30,    // Ohm
  R4: 40,    // Ohm
  U1: 5,     // Volt
  U2: 12     // Volt
});

console.log(result);
// -> { I1: '0.25 A', I2: '0.35 A', I3: '0.30 A', I4: '0.225 A' }

module.exports = solveCurrents;

  • Lesbarkeit: Ergebnisse kommen als Strings mit Einheit ("0.25 A").
  • Flexibilität: Du kannst später auch komplexere Schaltungen oder andere Einheiten (mA, kΩ) einbauen.

#!/usr/bin/env node

// currents-cli.js
// CLI-Tool zur Berechnung der Ströme I1, I2, I3, I4

function solveCurrents({ Ri1, Ri2, R3, R4, U1, U2 }) {
  const numerator = U1 / Ri1 + U2 / Ri2;
  const denominator = R3 / Ri1 + R3 / Ri2 + 1 + R3 / R4;
  const I3 = numerator / denominator;

  const I4 = (R3 / R4) * I3;
  const I2 = (U2 - R3 * I3) / Ri2;
  const I1 = (U1 - R3 * I3) / Ri1;

  return { I1, I2, I3, I4 };
}

// CLI-Argumente einlesen
// Beispiel: node currents-cli.js 10 20 30 40 5 12
const [Ri1, Ri2, R3, R4, U1, U2] = process.argv.slice(2).map(Number);

if ([Ri1, Ri2, R3, R4, U1, U2].some(isNaN)) {
  console.error("Bitte Zahlen eingeben: Ri1 Ri2 R3 R4 U1 U2");

const result = solveCurrents({ Ri1, Ri2, R3, R4, U1, U2 });

console.log(`I1 = ${result.I1.toFixed(3)} A`);

-- currents-ratio.hs
-- Variante mit exakten Brüchen

import Data.Ratio

-- Eingaben als Rational
solveCurrentsRatio :: Rational -> Rational -> Rational -> Rational -> Rational -> Rational -> (Rational, Rational, Rational, Rational)
solveCurrentsRatio ri1 ri2 r3 r4 u1 u2 =
  let numerator   = u1 / ri1 + u2 / ri2
      denominator = r3 / ri1 + r3 / ri2 + 1 + r3 / r4
      i3 = numerator / denominator
      i4 = (r3 / r4) * i3
      i2 = (u2 - r3 * i3) / ri2
      i1 = (u1 - r3 * i3) / ri1
  in (i1, i2, i3, i4)

-- Beispiel: solveCurrentsRatio (10%1) (20%1) (30%1) (40%1) (5%1) (12%1)
-- Ergebnis: (1 % 4, 7 % 20, 3 % 10, 9 % 40)

Hier bekommst du exakte Brüche wie 1 % 4 statt 0.25.
